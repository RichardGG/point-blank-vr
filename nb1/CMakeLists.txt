cmake_minimum_required(VERSION 3.16)
project(nb1 C)
set(CMAKE_C_STANDARD 11)
add_library(nb1 STATIC nb1_video.c nb1_scene.c nb1_json.c)
target_include_directories(nb1 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(nb1 PRIVATE _GNU_SOURCE)
find_library(M_LIB m)
if(M_LIB)
  target_link_libraries(nb1 PUBLIC ${M_LIB})
endif()
add_executable(nb1_tool nb1_tool.c)
target_link_libraries(nb1_tool nb1)
